### Validate pseudogene candidates using read mapping and variant calls with remap_to_coverage_variants.sh, which will:
1. Run remap of the MAG (Prokka output .ffn) agains the source metagenome using Bowtie2 and samtools
2. Run variant calling with bcftools
3. get read coverage to confirm disruptions in pseudogene candidates by analyzing read depth and alignment consistency.
4. analyze variants to identify mutations (e.g., frameshifts, premature stops) that support pseudogenization.

./tools/remap_coverage_variants.sh      \
        --ffn_file input/prokka_out/My_MAG.ffn     \
        --metagenome_file path/to/metagenome        \
        --prefix My_MAG

### Compare the pseudogene candidates with the original reference genome using blastn (genomic sequences)
### Analyze Alignment Results (only candidate pseudogenes with a problematic blast hit will be kept)
Compare the alignment results to identify:
  -Truncated proteins (alignment length < 70–80% of reference).
  -Frameshifts or internal stop codons (Look for high mismatch or gapopen values).
  -Percent identity (High identity suggests conserved regions).
  -summarize_parse_blast_output.py: Parse the BLAST output to identify pseudogenes with significant disruptions (e.g., truncated alignments, frameshifts).

./tools/compare_pseudogenes_with_reference.sh -h
./tools/compare_pseudogenes_with_reference.sh   \

##########################################################################################################################################################################################################################
PART 4 - Anvi'o metabolism, MAG workflow
##########################################################################################################################################################################################################################
# rename fasta headers
anvi-script-reformat-fasta My_MAG.fa -o contigs_clean.fa --simplify-names --report name_conversions_contigs.txt

# create anvio database
anvi-gen-contigs-database -f contigs_clean.fa -o My_MAG.db -T 32

# extract CDS protein sequences to run emapper
anvi-get-sequences-for-gene-calls -c My_MAG.db --get-aa-sequences -o My_MAG_anvio.faa

### run_emapper.job
emapper.py --cpu 32     \
        -m diamond      \
        -i My_MAG_anvio.faa        \
        -o My_MAG_anvio    \
        --output_dir ./emapper2anvio    \
        --report_orthologs      \
        --decorate_gff yes

### remap2anvio_profile.job
bowtie2-build contigs_clean.fa contigs_clean.fa

bowtie2 -p 32 -x contigs_clean.fa \
        -1 /path/to/metagenome/My_metagenome_R1.fastq      \
        -2 /path/to/metagenome/My_metagenome_R2.fastq      \
        -S remap2anvio_profile.sam
samtools view -@ 32 -b remap2anvio_profile.sam > remap2anvio_profile.bam
samtools sort -@ 32 remap2anvio_profile.bam > remap2anvio_profile_sorted.bam
samtools index -@ 32 remap2anvio_profile_sorted.bam
rm remap2anvio_profile.sam remap2anvio_profile.bam

# run anvi profile to upload remap data
anvi-profile -i remap2anvio_profile_sorted.bam   \
        -c My_MAG.db -o My_MAG_profile -T 32

# add a default collection to been able to get coverage in estimate metabolism step
anvi-script-add-default-collection -c My_MAG.db -p My_MAG_profile/PROFILE.db -b My_MAG

# upload emapper annotations
anvi-script-run-eggnog-mapper -c My_MAG.db --annotation ./emapper2anvio/My_MAG_anvio.emapper.annotations2anvio --use-version 2.1.9

# run taxonomic and functional annotations
anvi-run-hmms -c My_MAG.db --also-scan-trnas -T 32
anvi-run-scg-taxonomy -c My_MAG.db -T 32
anvi-estimate-scg-taxonomy -c My_MAG.db -T 32      \
        -o My_MAG_taxonomy.txt     \
        --report-scg-frequencies ./My_MAG_scg_freqs.tsv
anvi-run-ncbi-cogs -c My_MAG.db -T 32
anvi-run-kegg-kofams -c My_MAG.db -T 32 --log-bitscores
anvi-estimate-metabolism -c My_MAG.db -p My_MAG_profile/PROFILE.db -C DEFAULT --add-copy-number --add-coverage
anvi-estimate-metabolism -c My_MAG.db -O My_MAG --output-modes hits
anvi-export-functions -c My_MAG.db -o My_MAG_all_functional_annotations.tsv
anvi-export-gene-coverage-and-detection -c My_MAG.db -p My_MAG_profile/PROFILE.db -O My_MAG
anvi-estimate-genome-completeness -c My_MAG.db -p My_MAG_profile/PROFILE.db -C DEFAULT -o My_MAG_completeness.txt
